Faster PC — التوثيق




src/
  version.py            # APP_VERSION / APP_ID / CONFIG_SCHEMA_VERSION
  config/settings.py    # SettingsManager: atomic writes, deep merge, schema migration, validation
  core/
    app_context.py      # AppContext: wires every service (usable headless)
    exclusions.py       # ProtectionManager: 3-layer process/service protection
    monitor.py          # PerformanceMonitor: samples, history, sustained thresholds
    health.py           # HealthMonitor: Event Log crashes, disks, health report
    notifications.py    # NotificationCenter (+ Level)
    journal.py          # ChangeJournal: record/undo with registered handlers
    cleaner.py          # SystemCleaner: symlink-safe deletes, size accounting
    optimizer.py        # SystemOptimizer: processes/services/startup + undo handlers
    tweaks.py           # PerformanceTweaks: registry + powercfg tweaks with undo
    repair.py           # SystemRepair: SFC/DISM/restore/DNS/chkdsk
    installer.py        # SystemInstaller: shortcuts, run-at-startup, Defender
    results.py          # ActionResult dataclass
  gui/
    app.py              # QApplication bootstrap, single instance, excepthook
    setup_wizard.py     # QWizard-based first-run setup
    login_window.py     # login + lockout
    main_window.py      # tabs, tray, task manager usage
    settings_window.py  # all settings
    dialogs.py          # confirm/protection/notifications/report/about
    widgets.py theme.py # shared widgets + palette/stylesheet
  utils/
    winapi.py           # silent subprocess, shortcuts, recycle bin, admin, sysinfo
    registry.py         # safe registry access (+ winreg_stub for non-Windows)
    logger.py paths.py auth.py workers.py
tests/                  # pytest unit tests (headless)
tools/ui_preview.py     # offscreen screenshots of every tab
