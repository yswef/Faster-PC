  gui/settings_window.py # SettingsWindow + admin user management
  gui/login_window.py    # LoginWindow (first-run + auth)
  utils/auth.py          # AuthManager: PBKDF2 local accounts
  utils/logger.py        # UILogHandler (thread-safe via Qt Signal), file logging
  utils/workers.py       # TaskWorker (QThread wrapper)
main.py                  # entry point
config.json              # persisted settings + accounts + feature flags
